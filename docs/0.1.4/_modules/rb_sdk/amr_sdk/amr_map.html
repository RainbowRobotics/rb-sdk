    async def _map_flow_manager_solver(self, robot_model: str, req_id: str, flow_manager_args: FlowManagerArgs | None = None):
        """
        [Map 관련 함수에서 사용되는 flow manager 처리 함수]
        - robot_model: 로봇 모델명
        - req_id: 요청 아이디
        - flow_manager_args: flow manager 인자
        """
        if flow_manager_args is None:
            return None

        done_event = self._zenoh_subs_done_event()
        state: dict[str, Any] = {
            "finished": False,
            "error": "",
            "last_seen": time.monotonic(),
        }

        def all_stop(*, flow_manager_args: FlowManagerArgs | None = None):
            """모든 프로세스 정지"""
            with suppress(Exception):
                self.zenoh_client.query_one("muscat/stop", payload={}, timeout=10)
            if flow_manager_args is not None:
                flow_manager_args.done()

        async def _on_status(*, obj_payload=None, unsubscribe=None, **_):
            if state["finished"] or obj_payload is None:
                return

            try:
                flow_manager_args.ctx.check_stop()
                state["last_seen"] = time.monotonic()

                obj = t_to_dict(obj_payload)

                if obj.get("mapState").get("mapId") != req_id:
                    raise RuntimeError("Map ID Mismatch")

                if obj.get("mapState").get("mapResult") == "success":
                    print("Map Flow Manager SOLVER DONE", flush=True)
                    flow_manager_args.done()
                    state["finished"] = True
                    if unsubscribe is not None:
                        unsubscribe()
                    done_event.set()
                elif obj.get("mapState").get("mapResult") == "fail":
                    raise RuntimeError("Map Load Fail")
                elif obj.get("mapState").get("mapResult") == "cancel":
                    raise RuntimeError("Map Load Cancel")
            except Exception as e:
                state["finished"] = True
                state["error"] = str(e)
                if unsubscribe is not None:
                    unsubscribe()
                done_event.set()

        async def _watchdog():
            timeout_sec = 2.0
            interval_sec = 0.5
            while not state["finished"]:
                await asyncio.sleep(interval_sec)
                if state["finished"]:
                    return
                if time.monotonic() - state["last_seen"] > timeout_sec:
                    state["finished"] = True
                    state["error"] = "Map Load Fail (Timeout)"
                    done_event.set()
                    return

        async with self._zenoh_sub_with_close(
            f"amr/{robot_model}/status",
            _on_status,
            flatbuffer_obj_t=StatusT,
            options=SubscribeOptions(
                dispatch="queue",
                overflow=OverflowPolicy.LATEST_ONLY,
                parse_dict_payload=False,
            ),
        ):
            watchdog_task = asyncio.create_task(_watchdog())
            try:
                await done_event.wait()
                if state["error"] != "":
                    raise RuntimeError(state["error"])
            except asyncio.CancelledError:
                pass
            except Exception as e:
                all_stop(flow_manager_args=flow_manager_args)
                raise RuntimeError(str(e)) from e
            finally:
                watchdog_task.cancel()
                try:
                    await watchdog_task
                except asyncio.CancelledError:
                    print("WATCHDOG CANCELLED", flush=True)

        return None

    def get_map_list(self, robot_model: str, req_id: str | None = None, flow_manager_args: FlowManagerArgs | None = None) -> ResponseMapListT:
        """
