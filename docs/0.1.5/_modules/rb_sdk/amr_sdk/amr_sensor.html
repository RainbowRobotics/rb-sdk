    def marker_scan_start(self, robot_model: str, flow_manager_args: FlowManagerArgs | None = None) -> ResponseMarkerScanStartT:

        result = self.zenoh_client.query_one(
            f"amr/{robot_model}/marker/scanStart",
            flatbuffer_res_T_class=ResponseMarkerScanStartT,
            flatbuffer_buf_size=2,
            timeout=10,
        )

        if result.get("err") is not None and result.get("err") != "":
            raise RuntimeError(f"Query Err : {result.get("err")}")
        if result["obj_payload"] is None:
            raise RuntimeError("Marker Scan Start failed: obj_payload is None")

        if flow_manager_args is not None:
            if result["dict_payload"].get("result") == "accept":
                flow_manager_args.done()
            else:
                raise RuntimeError(result["dict_payload"].get("message"))

        return result.get("obj_payload")

    def rtsp_subscribe(self, robot_model: str, cam_id: int, quality: str) -> ResponseRTSPSubscribeT:
        """
        [RTSP 구독]
        - robot_model: 로봇 모델
        - cam_id: 카메라 ID
        - quality: 비디오 품질
        - ResponseRTSPSubscribeT 객체 반환
        """
        req = RequestRTSPSubscribeT()
        req.camId = cam_id
        req.quality = quality

        result = self.zenoh_client.query_one(
            f"amr/{robot_model}/sensor/rtsp/subscribe",
            flatbuffer_req_obj=req,
            flatbuffer_res_T_class=ResponseRTSPSubscribeT,
            flatbuffer_buf_size=125,
            timeout=10,
        )

        if result.get("err") is not None and result.get("err") != "":
            raise RuntimeError(f"Query Err : {result.get("err")}")
        if result.get("obj_payload") is None:
            raise RuntimeError("Call RTSP Subscribe failed: obj_payload is None")

        return result.get("obj_payload")

    def set_safety_lidar_mode(
        self,
        robot_model: str,
        switch: bool,
        flow_manager_args: FlowManagerArgs | None = None,
    ) -> ResponseSetSafetyLidarT:
        """
        [탐지 모드 제어]
        - req_id: 요청 ID
        - switch: 탐지 모드 켜기/끄기 (True: 켜기, False: 끄기)
        - ResponseSetSafetyLidarT 객체 반환
        """

        # 1) RequestSetDetectModeT 객체 생성
        req = RequestSetSafetyLidarT()
        req.switch = switch

        # 2) 요청 전송
        result = self.zenoh_client.query_one(
            f"amr/{robot_model}/sensor/setSafetyLidar",
            flatbuffer_req_obj=req,
            flatbuffer_res_T_class=ResponseSetSafetyLidarT,
            flatbuffer_buf_size=125,
            timeout=10,
        )

        if result.get("err") is not None and result.get("err") != "":
            raise RuntimeError(f"Query Err : {result.get("err")}")
        # 3) 결과 처리 및 반환
        if result.get("obj_payload") is None:
            raise RuntimeError("Call Control Motor failed: obj_payload is None")

        if flow_manager_args is not None:
            if result["dict_payload"].get("result") == "accept":
                flow_manager_args.done()
            else:
                raise RuntimeError(result["dict_payload"].get("message"))

        return result["obj_payload"]
