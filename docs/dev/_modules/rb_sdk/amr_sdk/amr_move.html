    async def _move_flow_manager_solver(
        self,
        *,
        robot_model: str,
        req_id: str,
        flow_manager_args: FlowManagerArgs | None = None,
    ):
        """
        [Move 관련 함수에서 사용되는 flow manager 처리 함수]

        Args:
            robot_model: 로봇 모델명
            req_id: 요청 아이디
            flow_manager_args: RB PFM을 쓸때 전달된 Flow Manager 인자 (done 콜백 등)
        """
        if flow_manager_args is None:
            return None

        done_event = self._zenoh_subs_done_event()
        state: dict[str, Any] = {
            "finished": False,
            "error": "",
            "last_seen": time.monotonic(),
        }


        def all_stop(*, flow_manager_args: FlowManagerArgs | None = None):
            """모든 프로세스 정지"""
            with suppress(Exception):
                self.zenoh_client.query_one("muscat/stop", payload={}, timeout=10)
            if flow_manager_args is not None:
                flow_manager_args.done()

        def _on_move_status(*, obj_payload=None, unsubscribe=None, **_):
            if state["finished"] or obj_payload is None:
                return

            try:
                state["last_seen"] = time.monotonic()
                flow_manager_args.ctx.check_stop()

                obj = t_to_dict(obj_payload)
                # print(f"FLOW MANAGER MOVESTATUS >>>>>>>>>>> {obj.get('moveState').get('moveId')} {req_id} {obj.get('moveState').get('moveResult')} ", flush=True)

                if obj.get("moveState").get("moveId") != req_id:
                    raise RuntimeError("Move ID Mismatch")

                if obj.get("moveState").get("moveResult") == "success":
                    # print("FLOW MANAGER SOLVER DONE", flush=True)
                    flow_manager_args.done()
                    state["finished"] = True
                    if unsubscribe is not None:
                        unsubscribe()
                    done_event.set()
                elif obj.get("moveState").get("moveResult") == "fail":
                    raise RuntimeError("Move Fail")
                elif obj.get("moveState").get("moveResult") == "cancel":
                    raise RuntimeError("Move Cancel")

            except Exception as e:
                state["finished"] = True
                state["error"] = str(e)
                if unsubscribe is not None:
                    unsubscribe()
                done_event.set()

        async def _watchdog():
            timeout_sec = 2.0
            interval_sec = 0.5
            while not state["finished"]:
                if time.monotonic() - state["last_seen"] - interval_sec > timeout_sec:
                    state["finished"] = True
                    state["error"] = "Move Fail (Timeout)"
                    done_event.set()
                    return

                await asyncio.sleep(interval_sec)

        async with self._zenoh_sub_with_close(
            f"amr/{robot_model}/moveStatus",
            _on_move_status,
            flatbuffer_obj_t=MoveStatusT,
            options=SubscribeOptions(
                dispatch="queue",
                overflow=OverflowPolicy.LATEST_ONLY,
                parse_dict_payload=False,
            ),
        ):
            watchdog_task = asyncio.create_task(_watchdog())
            try:
                await done_event.wait()
                if state["error"] != "":
                    raise RuntimeError(state["error"])
            except asyncio.CancelledError:
                pass
            except Exception as e:
                all_stop(flow_manager_args=flow_manager_args)
                raise RuntimeError(str(e)) from e
            finally:
                watchdog_task.cancel()
                try:
                    await watchdog_task
                except asyncio.CancelledError:
                    print("WATCHDOG CANCELLED", flush=True)
        return None

    def send_move_goal(
        self,
        robot_model: str,
        goal_id: str | None,
        goal_name: str | None,
        method: str | None = None,
        preset: int = 0,
        req_id: str | None = None,
        flow_manager_args: FlowManagerArgs | None = None,
    ) -> ResponseMoveGoalT:
        """
        [목표 위치로 이동]
        - robot_model: 로봇 모델명
        - req_id: 요청 아이디 (None인 경우 자동 생성)
        - goal_id: 목표 노드 ID
        - goal_name: 목표 노드 이름
        - method: 이동 방법 ( "pp", "hpp" )
        - preset: 프리셋 번호 (현재는 0으로 고정)
        - ResponseMoveGoalT 객체 반환
        """

        if req_id is None:
            req_id = str(uuid.uuid4())
        # 1) Request_Move_GoalT 객체 생성
        req = RequestMoveGoalT()
        req.id = req_id
        req.goalId = goal_id
        req.goalName = goal_name
        req.method = method
        req.preset = preset

        # 2) 요청 전송
        print("[AmrMoveSDK] Move Goal Request : ", robot_model, req_id, goal_id, goal_name, method, preset, flush=True)

        result = self.zenoh_client.query_one(
            f"amr/{robot_model}/move/goal",
            flatbuffer_req_obj=req,
            flatbuffer_res_T_class=ResponseMoveGoalT,
            flatbuffer_buf_size=125,
            timeout=10,
        )

        if result.get("err") is not None and result.get("err") != "":
            raise RuntimeError(f"Query Err : {result.get("err")}")
        # 3) 결과 처리 및 반환
        if result.get("obj_payload") is None:
            raise RuntimeError("Call Move Target failed: obj_payload is None")

        print("[AmrMoveSDK] Move Goal Response : ", result["dict_payload"].get("result"), result["dict_payload"].get("message"), flush=True)

        if flow_manager_args is not None:
            # variable_name = flow_manager_args.args.get("variable_name", None)
            # print(">>>>>>> variable Name : ",variable_name,flush=True)
            # print(result["dict_payload"].get("result"))
            # flow_manager_args.ctx.update_local_variables({"AMR_TE33ST": result["dict_payload"].get("result")})
            # flow_manager_args.done()
            if result["dict_payload"].get("result") == "reject":
                raise RuntimeError(result["dict_payload"].get("message"))

            self._run_coro_blocking(
                self._move_flow_manager_solver(
                    robot_model=robot_model,
                    req_id=req_id,
                    flow_manager_args=flow_manager_args,
                )
            )

        return result["obj_payload"]

    def send_move_target(self, robot_model: str,
    x: float, y: float, z: float, rz: float, method: str | None = None, preset: int = 0,
    req_id: str | None = None,
    flow_manager_args: FlowManagerArgs | None = None) -> ResponseMoveTargetT:
        """
        [특정 위치로 이동]

        goal_pose.rz = rz

        print("[AmrMoveSDK] Move Target Request : ", robot_model, req_id, x, y, z, rz, method, preset, flush=True)

        # 1) Request_Move_GoalT 객체 생성
        req = RequestMoveTargetT()
        req.id = req_id
        req.goalPose = goal_pose
        req.method = method
        req.preset = preset

        # 2) 요청 전송

            timeout=10,

            if result["dict_payload"].get("result") == "reject":

                self._move_flow_manager_solver(
                    robot_model=robot_model,
