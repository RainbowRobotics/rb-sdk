    async def _dock_flow_manager_solver(
        self,
        robot_model: str,
        req_id: str | None = None,
        flow_manager_args: FlowManagerArgs | None = None,
    ):
        """
        [Dock 관련 함수에서 사용되는 flow manager 처리 함수]
        """

        if flow_manager_args is None:
            return None

        if req_id is None:
            req_id = str(uuid.uuid4())

        done_event = self._zenoh_subs_done_event()
        state: dict[str, Any] = {
            "finished": False,
            "error": "",
            "last_seen": time.monotonic(),
        }

        def all_stop(*, flow_manager_args: FlowManagerArgs | None = None):
            """모든 프로세스 정지"""
            with suppress(Exception):
                self.zenoh_client.query_one("muscat/stop", payload={}, timeout=10)
            if flow_manager_args is not None:
                flow_manager_args.done()

        async def _on_status(*, obj_payload=None, unsubscribe=None, **_):
            if state["finished"] or obj_payload is None:
                return

            try:
                state["last_seen"] = time.monotonic()
                flow_manager_args.ctx.check_stop()

                obj = t_to_dict(obj_payload)
                print(
                    f"FLOW MANAGER STATUS >>>>>>>>>>> {obj.get('chargeState').get('dockId')} {req_id} {obj.get('chargeState').get('dockResult')} ",
                    flush=True,
                )

                if obj.get("chargeState").get("dockId") != req_id:
                    raise RuntimeError("Dock ID Mismatch")

                if obj.get("chargeState").get("dockResult") == "success":
                    print("Dock Flow Manager SOLVER DONE", flush=True)
                    flow_manager_args.done()
                    state["finished"] = True
                    if unsubscribe is not None:
                        unsubscribe()
                    done_event.set()
                elif obj.get("chargeState").get("dockResult") == "fail":
                    raise RuntimeError("Dock Fail")
                elif obj.get("chargeState").get("dockResult") == "cancel":
                    raise RuntimeError("Dock Cancel")
            except Exception as e:
                state["finished"] = True
                state["error"] = str(e)
                if unsubscribe is not None:
                    unsubscribe()
                done_event.set()

        async def _watchdog():
            timeout_sec = 10.0
            interval_sec = 0.5
            while not state["finished"]:
                await asyncio.sleep(interval_sec)
                if state["finished"]:
                    return
                if time.monotonic() - state["last_seen"] > timeout_sec:
                    state["finished"] = True
                    state["error"] = "Dock Fail (Timeout)"
                    done_event.set()
                    return

        async with self._zenoh_sub_with_close(
            f"amr/{robot_model}/status",
            _on_status,
            flatbuffer_obj_t=StatusT,
            options=SubscribeOptions(
                dispatch="queue",
                overflow=OverflowPolicy.LATEST_ONLY,
                parse_dict_payload=False,
            ),
        ):
            watchdog_task = asyncio.create_task(_watchdog())
            try:
                await done_event.wait()
                if state["error"] != "":
                    raise RuntimeError(state["error"])
            except asyncio.CancelledError:
                pass
            except Exception as e:
                all_stop(flow_manager_args=flow_manager_args)
                raise RuntimeError(str(e)) from e
            finally:
                watchdog_task.cancel()
                try:
                    await watchdog_task
                except asyncio.CancelledError:
                    print("WATCHDOG CANCELLED", flush=True)

        return None

    def control_get_safety_field(
        self,
        robot_model: str,
        req_id: str | None = None,
        flow_manager_args: FlowManagerArgs | None = None,
    ) -> ResponseGetSafetyFieldT:
        """
        [안전 필드 조회]
        - ResponseGetSafetyFieldT 객체 반환
        """

        if req_id is None:
            req_id = str(uuid.uuid4())

        # 1) RequestGetSafetyFieldT 객체 생성
        req = RequestGetSafetyFieldT()
        req.id = req_id

        print("[AmrControlSDK] getSafetyField : ", req_id, flush=True)

        # 2) 요청 전송
        result = self.zenoh_client.query_one(
            f"amr/{robot_model}/control/getSafetyField",
            flatbuffer_req_obj=req,
            flatbuffer_res_T_class=ResponseGetSafetyFieldT,
            flatbuffer_buf_size=125,
            timeout=10,
        )

        # 3) 결과 처리 및 반환

            raise RuntimeError("Call Control Safety Field failed: obj_payload is None")

        if flow_manager_args is not None:
            if result["dict_payload"].get("result") == "accept":
                flow_manager_args.done()
            else:
                raise RuntimeError(result["dict_payload"].get("message"))

        return result["obj_payload"]
