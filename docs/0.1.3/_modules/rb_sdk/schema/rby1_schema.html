rb_sdk.schema.rby1_schema의 소스 코드

"""RB-Y1 SDK 스키마 (TypedDict).

공식 ``rby1_sdk`` 의 pybind 구조체(RobotState_* 등)를 순수 dict 로 변환할 때의
스키마. SDK 레이어는 런타임 오버헤드 0 인 **TypedDict** 만 노출한다.
Pydantic 모델이 필요하면 services 레이어에서 이 TypedDict 로 파생한다.

변환 엔진은 ``rb_sdk.rby1_sdk.base.RBRby1BaseSDK.struct_to_dict`` 참조.
  - 관절 벡터는 numpy ndarray → list[float], enum → str(name), datetime 유지
  - A/M/UB가 동일 필드를 공유. DOF = robot.model().robot_dof (A=24, M=23, UB=18)
  - total=False: pybind 객체에서 일부 필드 추출 실패해도 부분 직렬화 허용
"""

from __future__ import annotations

from datetime import datetime
from typing import Any, TypedDict

class SystemStatTD(TypedDict, total=False):

    memory_usage: float | None
    uptime: float | None
    program_uptime: float | None

class BatteryStateTD(TypedDict, total=False):

    current: float | None
    level_percent: float | None

class PowerStateTD(TypedDict, total=False):

    voltage: float | None

class EMOStateTD(TypedDict, total=False):

class FTSensorDataTD(TypedDict, total=False):

    torque: list[float] | None  # (3,) Tx, Ty, Tz [Nm]
    time_since_last_update: float | None

    digital_output_A: bool | None
    digital_output_B: bool | None

class RobotStateTD(TypedDict, total=False):
    """RB-Y1 RobotState (한 시점 전체 스냅샷)."""

    timestamp: datetime | None

    # 관절 벡터 (shape (DOF,))
    is_ready: list[bool] | None
    position: list[float] | None  # [rad]
    velocity: list[float] | None  # [rad/s]
    current: list[float] | None  # [A]
    torque: list[float] | None  # [Nm]
    temperature: list[float] | None  # [°C]
    gravity: list[float] | None  # 중력보상 토크 [Nm]
    target_position: list[float] | None  # [rad]
    target_velocity: list[float] | None  # [rad/s]
    target_feedback_gain: list[float] | None  # [0, 10]
    target_feedforward_torque: list[float] | None  # [Nm]

    # 베이스/기타
    odometry: list[list[float]] | None  # SE(2) 3x3 동차변환
    center_of_mass: list[float] | None  # (3,) base frame [m]
    collisions: list[dict[str, Any]] | None

    # 중첩 구조체
    system_stat: SystemStatTD | None
    battery_state: BatteryStateTD | None
    power_states: list[PowerStateTD] | None
    emo_states: list[EMOStateTD] | None
    joint_states: list[JointStateTD] | None
    tool_flange_left: ToolFlangeStateTD | None
    tool_flange_right: ToolFlangeStateTD | None
    ft_sensor_left: FTSensorDataTD | None
    ft_sensor_right: FTSensorDataTD | None
