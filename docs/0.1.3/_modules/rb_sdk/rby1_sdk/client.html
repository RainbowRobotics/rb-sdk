from __future__ import annotations

import contextlib
import json
import re
from typing import Any

import rby1_sdk
from rb_log.log import rb_log

from .config import RBY1Config
from .exceptions import RBY1SDKConnectionError, RBY1SDKError

# robot.get_last_log() 는 폴링이라 실패 직후 딱 그 순간의 로그를 못 잡을 수 있다
# (라이브 확인: Debug 로그가 API 호출마다 다량으로 섞여 나옴). start_log_stream 은
# 로봇 쪽에서 실시간으로 밀어주는 방식이라 놓치지 않는다 -- Warn 이상만 골라
# muscat/log/write 로 relay 한다(Trace/Debug/Info 까지 다 보내면 너무 시끄럽다).
_LOG_STREAM_RATE_HZ = 5.0
_FORWARD_LEVELS = {"Warn", "Error", "Critical"}

# rby1_sdk 로그 원문 맨 앞의 "[HH:MM:SS.ffffff]" 자체 타임스탬프 -- 로그 뷰어의
# Created At 컬럼과 중복이라 빼고 보낸다(컴포넌트 태그 "[ControlManager::...]"는
# 다른 곳엔 없는 정보라 유지).
_TIMESTAMP_PREFIX_RE = re.compile(r"^\[\d{2}:\d{2}:\d{2}(?:\.\d+)?\]\s*")

def extract_log_message(raw: str) -> str:
    """서버 로그 한 줄에서 사람이 읽을 사유만 뽑아낸다.

    rby1_sdk 의 검증 실패 로그는 ``"[timestamp] [Component] Invalid request: {...중첩 JSON...}"``
    형태고, 실제 이유는 그 JSON 맨 안쪽 ``control.control...control.message`` 에
    깊이 묻혀 있다(라이브 확인 — nesting 깊이는 어느 컴포넌트/커맨드 타입이냐에
    따라 달라진다). 로그 뷰어에 그 JSON 덩어리를 통째로 보여주면 못 읽으므로,
    JSON 을 파싱해서 가장 안쪽 ``message`` 키만 재귀적으로 찾아 반환한다.
    JSON 이 없는 일반 로그(예: ``ControlManager`` 상태 메시지)는 맨 앞 자체
    타임스탬프만 떼고 나머지는 그대로 돌려준다.
    """
    idx = raw.find("{")
    if idx == -1:
        return _TIMESTAMP_PREFIX_RE.sub("", raw, count=1)
    try:
        payload = json.loads(raw[idx:])
    except (ValueError, TypeError):
        return _TIMESTAMP_PREFIX_RE.sub("", raw, count=1)
    return _find_message_key(payload) or _TIMESTAMP_PREFIX_RE.sub("", raw, count=1)




def _find_message_key(node: Any) -> str | None:
    if isinstance(node, dict):
        message = node.get("message")
        if isinstance(message, str):
            return message
        for value in node.values():
            found = _find_message_key(value)
            if found is not None:
                return found
    elif isinstance(node, list):
        for item in node:
            found = _find_message_key(item)
            if found is not None:
                return found
    return None

class RBY1ClientSDK:
    """Thin adapter over official ``rby1_sdk``."""

    def __init__(self, config: RBY1Config):
        self.config = config
        self._robot: Any | None = None
        self._last_log_ts: Any | None = None

    @property
    def robot(self) -> Any | None:
        return self._robot

    @property
    def connected(self) -> bool:
        if self._robot is None:
            return False
        try:
            return bool(self._robot.is_connected())
        except Exception:
            return False

    def connect(self) -> Any:
        if self.config.model is not None:
            robot = rby1_sdk.create_robot(self.config.endpoint, self.config.model)
        else:
            robot = rby1_sdk.create_robot(self.config.endpoint)

        ok = robot.connect()
        if ok is False:
            raise RBY1SDKConnectionError(f"failed_to_connect_rby1:{self.config.endpoint}")

        self._robot = robot
        with contextlib.suppress(Exception):
            robot.start_log_stream(self._on_robot_log, _LOG_STREAM_RATE_HZ)
        return robot

    def disconnect(self):
        if self._robot is None:
            return
        with contextlib.suppress(Exception):
            self._robot.stop_log_stream()
        with contextlib.suppress(Exception):
            self._robot.disconnect()
        self._robot = None



    def _on_robot_log(self, entries: list[Any]) -> None:
        # start_log_stream 콜백은 매 tick 마다 "최근 로그" 창을 통째로 넘겨주고
        # 창이 겹치므로(라이브 확인: 같은 항목이 여러 tick 에 걸쳐 반복 수신됨),
        # 이미 relay 한 항목보다 새 timestamp 인 것만 내보낸다.
        model = self.config.model or self.config.endpoint
        for entry in entries:
            ts = entry.robot_system_timestamp
            if self._last_log_ts is not None and ts <= self._last_log_ts:
                continue
            self._last_log_ts = ts

            level_name = entry.level.name
            if level_name not in _FORWARD_LEVELS:
                continue
            message = extract_log_message(entry.message)
            if level_name == "Warn":
                rb_log.warning(message, sw_name=model)
            else:
                rb_log.error(message, sw_name=model)

    def whoami(self) -> dict[str, Any]:
        if self._robot is None:
            raise RBY1SDKConnectionError("rby1_not_connected")

        result = self._robot.get_robot_info()
        if isinstance(result, dict):
            return result
        if hasattr(result, "__dict__"):
            return dict(vars(result))
        return {"result": str(result)}

    def call(self, method: str, *args: Any, **kwargs: Any) -> Any:
        if self._robot is None:
            raise RBY1SDKConnectionError("rby1_not_connected")

        fn = getattr(self._robot, method, None)
        if fn is None:
            raise RBY1SDKError(f"rby1_method_not_found:{method}")
        return fn(*args, **kwargs)
