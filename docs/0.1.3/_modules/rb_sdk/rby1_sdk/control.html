    def _call_existing(
        self,
        robot_model: str,
        *method_names: str,
        **kwargs: Any,
    ) -> Any:
        """``robot``에서 후보 method 중 첫 매칭 실행 — ``asyncio.to_thread`` 안에서만 호출할 것."""
        robot = self.get_robot(robot_model)
        if robot is None:
            raise RuntimeError(f"rby1_not_connected: {robot_model}")

        for method in method_names:
            fn = getattr(robot, method, None)
            if callable(fn):
                return fn(**kwargs)

        raise RuntimeError(
            f"rby1 methods not found on {robot_model}: {method_names}"
        )

        self,
        *,
        robot_model: str,
        method: str,
        args: tuple[Any, ...] = (),
        kwargs: dict[str, Any] | None = None,
        flow_manager_args: FlowManagerArgs | None = None,
    ) -> Any:
        """[RB-Y1 원본 제어 메서드 호출]"""
        client = self.get_client(robot_model)
        result = await asyncio.to_thread(client.call, method, *args, **(kwargs or {}))
        self._finish_flow(flow_manager_args=flow_manager_args)
        return result

    def _sync_stop(self, robot_model: str, priority: int) -> dict[str, Any]:
        """``StopCommandBuilder``로 정지 — 공식 SDK엔 stop()/Stop() 없음(확인됨), 이 경로가 유일."""
        import rby1_sdk as rby

        robot = self.get_robot(robot_model)
        if robot is None:
            raise RuntimeError(f"rby1_not_connected: {robot_model}")

        command = rby.RobotCommandBuilder().set_command(
            rby.WholeBodyCommandBuilder().set_command(rby.StopCommandBuilder())
        )
        handler = robot.send_command(builder=command, priority=priority)
        feedback = handler.get()
        finish_code = self._check_finish_code(robot, feedback, robot_model)
        return {"finished": handler.is_done(), "finish_code": finish_code}

    def _sync_emergency_stop(self, robot_model: str) -> dict[str, Any]:
        """모션 정지(best-effort) 후 무조건 전원 차단 — stop 실패해도 차단은 계속 시도(최후 안전장치)."""
        robot = self.get_robot(robot_model)
        if robot is None:
            raise RuntimeError(f"rby1_not_connected: {robot_model}")

        with contextlib.suppress(Exception):
            self._sync_stop(robot_model, priority=10)

        powered_off = robot.power_off(dev_name=".*")
        return {"stopped": bool(powered_off)}

    @staticmethod
    def _find_out_of_range_targets(robot: Any, *, margin_deg: float = 5.0) -> dict[str, float]:
        """리밋 벗어난 관절을 자기 리밋 안쪽(margin)으로 되돌릴 목표각을 계산 — enable 거부 시 자동복구용.

        ``get_dynamics().get_joint_names()``(해시맵 순서)와 ``joint_infos``(DOF 순서)가
        서로 달라 인덱스로 안 엮고 이름으로만 연결한다. 바퀴(연속회전)는 대상에서 제외.
        """
        import math

        dyn = robot.get_dynamics()
        dyn_names = list(dyn.get_joint_names())
        st = dyn.make_state(dyn.get_link_names(), dyn_names)
        q_lo = list(dyn.get_limit_q_lower(st))
        q_hi = list(dyn.get_limit_q_upper(st))
        limits_by_name = dict(zip(dyn_names, zip(q_lo, q_hi, strict=True), strict=True))

        info = robot.get_robot_info()
        info_names = [ji.name for ji in info.joint_infos]
        position_by_name = dict(zip(info_names, robot.get_state().position, strict=True))

        margin = math.radians(margin_deg)
        targets: dict[str, float] = {}
        for name, (lo, hi) in limits_by_name.items():
            if lo <= -1.0e6 or hi >= 1.0e6:
                continue  # 바퀴 등 연속회전 관절 -- 대상 아님
            pos = position_by_name.get(name)
            if pos is None or lo < pos < hi:
                continue
            safe_margin = min(margin, (hi - lo) / 2)
            target = lo + safe_margin if pos <= lo else hi - safe_margin
            targets[name] = math.degrees(target)
        return targets

    def _ensure_control_manager_ready(self, robot: Any, robot_model: str) -> None:
        """Enabled면 통과, Fault/Idle이면 자동 reset+enable — move 호출자가 매번 손으로 안 해도 되게.

        enable_control_manager 는 실패해도 예외 없이 조용히 Idle 로 남는다(예: 관절이
        리밋 밖이면 거부) — 그대로 move 하면 애매한 에러로 나오므로 여기서 실제
        Enabled 여부를 확인한다. 실패했는데 리밋 벗어난 관절이 있으면 unlimited_mode로
        강제 enable → 그 관절만 복귀 → 정상 enable 재시도(실측된 반복 케이스), 없으면 에러.
        """
        if robot.get_control_manager_state().state.name == "Enabled":
            return
        robot.reset_fault_control_manager()
        robot.enable_control_manager(unlimited_mode_enabled=False)

        if robot.get_control_manager_state().state.name == "Enabled":
            return

        targets = self._find_out_of_range_targets(robot)
        if not targets:
            state_after = robot.get_control_manager_state().state.name
            raise RuntimeError(
                f"rby1_control_manager_enable_failed: robot_model={robot_model} state={state_after}"
            )

        robot.reset_fault_control_manager()
        robot.enable_control_manager(unlimited_mode_enabled=True)
        self._sync_move_joints(robot_model, targets, 2.0, 10, relative=False)

        robot.reset_fault_control_manager()
        robot.disable_control_manager()
        robot.enable_control_manager(unlimited_mode_enabled=False)

        state_after = robot.get_control_manager_state().state.name
        if state_after != "Enabled":
            raise RuntimeError(
                f"rby1_control_manager_enable_failed_after_recovery: robot_model={robot_model} "
                f"state={state_after} recovered_joints={sorted(targets)}"
            )

    def _check_finish_code(self, robot: Any, feedback: Any, robot_model: str) -> str:
        """finish_code 확인 — is_done()/status 는 Canceled/Preempted 에도 Finished 로 뜨므로 그것만으론 성공 여부를 알 수 없다.

        FinishCode enum 엔 "요청 검증 거부"에 대응하는 값이 없어 그 경우 기본값
        Unknown 으로 온다(확인됨) — 예외 대신 그대로 반환하고, 실패 시 알림을 보낸다.
        """
        code = feedback.finish_code
        if code.name != "Ok":
            self._publish_move_error(robot, robot_model, code.name)
        return code.name

    def _publish_move_error(self, robot: Any, robot_model: str, finish_code_name: str) -> None:
        """실패한 move/stop 명령을 ``muscat/{robot_model}/message`` 로 알린다."""
        from rb_flat_buffers.muscat.Muscat_Message import Muscat_MessageT
        from rb_flat_buffers.muscat.Muscat_Message_Type import Muscat_Message_Type

        detail = ""
        with contextlib.suppress(Exception):
            logs = robot.get_last_log(1)
            if logs:
                detail = extract_log_message(logs[-1].message)

        msg = Muscat_MessageT()
        msg.type = Muscat_Message_Type.ERROR
        msg.isToast = False
        msg.strCode = f"RBY1 Move Error: {finish_code_name}"
        msg.subStr = f"{detail}" if detail else finish_code_name

        with contextlib.suppress(Exception):
            self.zenoh_client.publish(
                f"muscat/{robot_model}/message", flatbuffer_req_obj=msg, flatbuffer_buf_size=512
            )

    def _sync_move_joint_position(
        self,
        robot_model: str,
        position: list[float],
        minimum_time: float,
        velocity_limit: list[float] | None,
        acceleration_limit: list[float] | None,
        priority: int,
    ) -> dict[str, Any]:
        """body(torso+양팔) 전체 위치 명령 조립+전송. position 길이는 body DOF 와 일치해야 함(예: model A = 20)."""
        import numpy as np
        import rby1_sdk as rby

        robot = self.get_robot(robot_model)
        if robot is None:
            raise RuntimeError(f"rby1_not_connected: {robot_model}")
        self._ensure_control_manager_ready(robot, robot_model)

        jp = rby.JointPositionCommandBuilder().set_position(
            np.asarray(position, dtype=float)
        )
        jp = jp.set_minimum_time(minimum_time)
        if velocity_limit is not None:
            jp = jp.set_velocity_limit(np.asarray(velocity_limit, dtype=float))
        if acceleration_limit is not None:
            jp = jp.set_acceleration_limit(np.asarray(acceleration_limit, dtype=float))

        command = rby.RobotCommandBuilder().set_command(
            rby.ComponentBasedCommandBuilder().set_body_command(
                rby.BodyCommandBuilder().set_command(jp)
            )
        )

        # 디스패치는 다른 메서드와 동일하게 _call_existing 으로 (get_robot + 미연결 체크 재사용)
        handler = self._call_existing(
            robot_model, "send_command", builder=command, priority=priority
        )
        feedback = handler.get()  # 모션 완료까지 blocking (이미 worker thread 라 루프 안 막음)
        finish_code = self._check_finish_code(robot, feedback, robot_model)
        return {"finished": handler.is_done(), "finish_code": finish_code}

    @staticmethod
    def _group_by_component(
        names: list[str],
        torso_joint_idx: list[int],
        right_arm_joint_idx: list[int],
        left_arm_joint_idx: list[int],
        head_joint_idx: list[int],
        values_deg: dict[str, float],
    ) -> dict[str, dict[str, float]]:
        """관절명 -> 컴포넌트(torso/right_arm/left_arm/head) 매핑. 바퀴는 대상 밖이라 안 들어오면 unsupported로 걸러짐."""
        component_of: dict[str, str] = {}
        for i in torso_joint_idx:
            component_of[names[i]] = "torso"
        for i in right_arm_joint_idx:
            component_of[names[i]] = "right_arm"
        for i in left_arm_joint_idx:
            component_of[names[i]] = "left_arm"
        for i in head_joint_idx:
            component_of[names[i]] = "head"

        unsupported = [n for n in values_deg if n not in component_of]
        if unsupported:
            raise ValueError(
                f"unsupported joint name(s) (only torso/right_arm/left_arm/head): {unsupported}"
            )

        grouped: dict[str, dict[str, float]] = {
            "torso": {},
            "right_arm": {},
            "left_arm": {},
            "head": {},
        }
        for name, value_deg in values_deg.items():
            grouped[component_of[name]][name] = value_deg
        return grouped

    @staticmethod
    def _compute_subset_target(
        names: list[str],
        position: list[float],
        touched: dict[str, float],
        *,
        relative: bool,
    ) -> tuple[list[str], list[float]]:
        """지정한 관절만 대상인 target(torso 전용) — 나머지 torso 관절은 명령에서 아예 빠짐."""
        import math

        joint_names = list(touched.keys())
        values = []
        for name in joint_names:
            angle = math.radians(touched[name])
            values.append(position[names.index(name)] + angle if relative else angle)
        return joint_names, values

    @staticmethod
    def _compute_full_vector(
        names: list[str],
        component_joint_idx: list[int],
        position: list[float],
        touched: dict[str, float],
        *,
        relative: bool,
    ) -> list[float]:
        """팔/head 전용 — 컴포넌트 전체 고정 길이 벡터만 받으므로 지정 안 한 관절은 현재값으로 채움(hold)."""
        import math

        vec: list[float] = []
        for i in component_joint_idx:
            name = names[i]
            if name in touched:
                angle = math.radians(touched[name])
                vec.append(position[i] + angle if relative else angle)
            else:
                vec.append(position[i])
        return vec

    def _sync_move_joints(
        self,
        robot_model: str,
        values_deg: dict[str, float],
        minimum_time: float,
        priority: int,
        *,
        relative: bool,
    ) -> dict[str, Any]:
        """지정한 여러 관절을 한 번에 이동(절대/상대). 안 건드린 컴포넌트는 커맨드에서 아예 뺀다 —
        body 20-DOF 벡터 하나로 뭉치면 안 건드린 컴포넌트까지 겹침 판정돼 다른 동시
        명령과 불필요하게 취소/interrupt 될 수 있어서다. torso 는 이름 지정 subset
        타겟팅, 팔/head 는 컴포넌트 전체 벡터 중 안 지정한 관절만 현재값 유지(hold)."""
        import numpy as np
        import rby1_sdk as rby

        robot = self.get_robot(robot_model)
        if robot is None:
            raise RuntimeError(f"rby1_not_connected: {robot_model}")
        self._ensure_control_manager_ready(robot, robot_model)

        info = robot.get_robot_info()
        names = [ji.name for ji in info.joint_infos]
        grouped = self._group_by_component(
            names,
            list(info.torso_joint_idx),
            list(info.right_arm_joint_idx),
            list(info.left_arm_joint_idx),
            list(info.head_joint_idx),
            values_deg,
        )

        if not any(grouped.values()):
            raise ValueError("no joints specified")

        position = list(robot.get_state().position)

        body_component = None
        if grouped["torso"]:
            joint_names, values = self._compute_subset_target(
                names, position, grouped["torso"], relative=relative
            )
            torso_group = (
                rby.JointGroupPositionCommandBuilder()
                .set_joint_names(joint_names)
                .set_position(np.asarray(values, dtype=float))
                .set_minimum_time(minimum_time)
            )
            body_component = rby.BodyComponentBasedCommandBuilder().set_torso_command(torso_group)
        for side, idx_attr, setter in (
            ("right_arm", "right_arm_joint_idx", "set_right_arm_command"),
            ("left_arm", "left_arm_joint_idx", "set_left_arm_command"),
        ):
            if grouped[side]:
                vec = self._compute_full_vector(
                    names, list(getattr(info, idx_attr)), position, grouped[side], relative=relative
                )
                arm_jp = (
                    rby.JointPositionCommandBuilder()
                    .set_position(np.asarray(vec, dtype=float))
                    .set_minimum_time(minimum_time)
                )
                body_component = body_component or rby.BodyComponentBasedCommandBuilder()
                body_component = getattr(body_component, setter)(arm_jp)

        comp = rby.ComponentBasedCommandBuilder()
        if body_component is not None:
            comp = comp.set_body_command(body_component)
        if grouped["head"]:
            vec = self._compute_full_vector(
                names, list(info.head_joint_idx), position, grouped["head"], relative=relative
            )
            head_jp = (
                rby.JointPositionCommandBuilder()
                .set_position(np.asarray(vec, dtype=float))
                .set_minimum_time(minimum_time)
            )
            comp = comp.set_head_command(head_jp)

        command = rby.RobotCommandBuilder().set_command(comp)
        handler = self._call_existing(
            robot_model, "send_command", builder=command, priority=priority
        )
        feedback = handler.get()
        finish_code = self._check_finish_code(robot, feedback, robot_model)
        return {"finished": handler.is_done(), "finish_code": finish_code}

    async def move_joints_relative(
        self,
        *,
        robot_model: str,
        deltas_deg: dict[str, float],
        minimum_time: float = 3.0,
        priority: int = 1,
        flow_manager_args: FlowManagerArgs | None = None,
    ) -> dict[str, Any]:
        """[RB-Y1 여러 관절 동시 상대 이동] 현재 각도 기준 ``{관절명: ±각도(deg)}``만큼 이동.

        예: ``deltas_deg={"torso_0": 5.0, "right_arm_3": -10.0}``. 컴포넌트 스코핑
        규칙은 ``_sync_move_joints`` 참고.
        """
        result = await asyncio.to_thread(
            self._sync_move_joints, robot_model, deltas_deg, minimum_time, priority, relative=True
        )
        self._finish_flow(flow_manager_args=flow_manager_args)
        return result
