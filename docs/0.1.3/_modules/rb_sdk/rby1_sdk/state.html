"""RB-Y1 State SDK.

모든 호출 메서드는 ``async def`` — 공식 ``rby1_sdk``가 sync gRPC stub이라
``asyncio.to_thread`` 로 worker thread에 위임한다. 자세한 이유는 ``README.md`` 참조.

스키마 정책:
    SDK 레이어는 **TypedDict** (런타임 오버헤드 0, 고주파 스트림에 적합)만 노출한다.
    Pydantic 모델이 필요하면 services 레이어에서 이 TypedDict로 파생한다
    (예: ``pydantic.create_model`` / ``TypeAdapter``). SDK는 Pydantic에 의존하지 않는다.
"""

from __future__ import annotations

import asyncio
from collections.abc import Callable
from typing import Any

from rb_schemas.sdk import FlowManagerArgs

from ..schema.rby1_schema import ControlManagerStateTD, RobotStateTD
from .base import RBRby1BaseSDK

    """Rainbow Robotics RB-Y1 State SDK"""

    #   staticmethod 라 __init_subclass__ 의 safe-handler 래핑에서 제외된다.

[문서]
    @staticmethod
    def robot_state_to_dict(s: Any) -> RobotStateTD:
        """공식 ``rby1_sdk`` ``RobotState_*`` 구조체 → ``RobotStateTD`` (순수 dict).

        이미 dict 면 그대로 반환 (테스트/호환성).
        """
        if isinstance(s, dict):
            return s  # type: ignore[return-value]
        return RBRby1BaseSDK.struct_to_dict(s, RobotStateTD)  # type: ignore[return-value]





[문서]
    @staticmethod
    def control_manager_state_to_dict(o: Any) -> ControlManagerStateTD | None:
        """``ControlManagerState`` → ``ControlManagerStateTD`` (start_state_stream 2번째 인자용)."""
        return RBRby1BaseSDK.struct_to_dict(o, ControlManagerStateTD)  # type: ignore[return-value]



    def _sync_get_state(self, robot_model: str) -> RobotStateTD:
        """worker thread에서 실행될 sync 호출 (``get_state`` 호환 메서드 fallback 포함)."""
        robot = self.get_robot(robot_model)
        if robot is None:
            raise RuntimeError(f"rby1_not_connected: {robot_model}")

        for method in ("get_state", "GetState", "state"):
            fn = getattr(robot, method, None)
            if callable(fn):
                return self.robot_state_to_dict(fn())

        raise RuntimeError(f"rby1 get_state not found on {robot_model}")



[문서]
    async def get_state(
        self,
        *,
        robot_model: str,
        flow_manager_args: FlowManagerArgs | None = None,
    ) -> RobotStateTD:
        """[RB-Y1 상태 조회] — 전체 RobotState 스냅샷 (TypedDict)."""
        result = await asyncio.to_thread(self._sync_get_state, robot_model)
        self._finish_flow(flow_manager_args=flow_manager_args)
        return result

    async def get_robot_info(
        self,
        *,
        robot_model: str,
        flow_manager_args: FlowManagerArgs | None = None,

        """[RB-Y1 robot info 조회] — ``robot.get_robot_info()``."""
        client = self.get_client(robot_model)
        result = await asyncio.to_thread(client.whoami)
        self._finish_flow(flow_manager_args=flow_manager_args)

    def _sync_get_control_manager_state(self, robot_model: str) -> ControlManagerStateTD | None:
        """worker thread 전용 — ``robot.get_control_manager_state()`` 원샷 조회."""
        robot = self.get_robot(robot_model)
        if robot is None:
            raise RuntimeError(f"rby1_not_connected: {robot_model}")
        return self.control_manager_state_to_dict(robot.get_control_manager_state())

        result = await asyncio.to_thread(self._sync_get_control_manager_state, robot_model)
        self._finish_flow(flow_manager_args=flow_manager_args)
        return result



    def _sync_get_joint_limits(self, robot_model: str) -> dict[str, Any]:
        """worker thread 전용 — dynamics 모델에서 관절별 리밋 조회."""
        robot = self.get_robot(robot_model)
        if robot is None:
            raise RuntimeError(f"rby1_not_connected: {robot_model}")

        dyn = robot.get_dynamics()
        names = list(dyn.get_joint_names())
        st = dyn.make_state(dyn.get_link_names(), names)

        q_lo = list(dyn.get_limit_q_lower(st))
        q_hi = list(dyn.get_limit_q_upper(st))
        v_lo = list(dyn.get_limit_qdot_lower(st))
        v_hi = list(dyn.get_limit_qdot_upper(st))
        a_lo = list(dyn.get_limit_qddot_lower(st))
        a_hi = list(dyn.get_limit_qddot_upper(st))
        tq = list(dyn.get_limit_torque(st))

        return {
            name: {
                "position": [q_lo[i], q_hi[i]],       # [lower, upper] rad
                "velocity": [v_lo[i], v_hi[i]],       # rad/s
                "acceleration": [a_lo[i], a_hi[i]],   # rad/s^2
                "torque": tq[i],                      # Nm
            }
            for i, name in enumerate(names)
        }

    async def get_joint_limits(
        self,
        *,
        robot_model: str,
        flow_manager_args: FlowManagerArgs | None = None,
    ) -> dict[str, Any]:
        """[RB-Y1 관절 리밋 조회] 관절명 → {position/velocity/acceleration [lower,upper], torque}.

        move_joint_position 전에 target 이 position 리밋 안에 드는지 검증하는 용도.
        (rby1 은 리밋 초과 시 에러 없이 해당 관절을 현재 위치에 hold 하므로, 사전 검증 필요.)
        wheel_* 은 연속회전이라 리밋이 ±float max 로 사실상 무한대.
        """
        result = await asyncio.to_thread(self._sync_get_joint_limits, robot_model)
        self._finish_flow(flow_manager_args=flow_manager_args)
        return result



    def _sync_start_state_stream(
        self,
        robot_model: str,
        on_state: Callable[..., Any],
        rate: float,
    ) -> None:
        """worker thread에서 실행될 sync 호출 (``start_state_update`` 호환 fallback 포함)."""
        robot = self.get_robot(robot_model)
        if robot is None:
            raise RuntimeError(f"rby1_not_connected: {robot_model}")

        for method in ("start_state_update", "StartStateUpdate"):
            fn = getattr(robot, method, None)
            if callable(fn):
                fn(on_state, rate)
                return

        raise RuntimeError(f"rby1 start_state_update not found on {robot_model}")

    def _sync_stop_state_stream(self, robot_model: str) -> None:
        """worker thread에서 실행될 sync 호출 (``stop_state_update`` 호환 fallback 포함)."""
        robot = self.get_robot(robot_model)
        if robot is None:
            raise RuntimeError(f"rby1_not_connected: {robot_model}")

        for method in ("stop_state_update", "StopStateUpdate"):
            fn = getattr(robot, method, None)
            if callable(fn):
                fn()
                return

        raise RuntimeError(f"rby1 stop_state_update not found on {robot_model}")



[문서]
    async def start_state_stream(
        self,
        *,
        robot_model: str,
        on_state: Callable[..., Any],
        rate: float = 100.0,
        flow_manager_args: FlowManagerArgs | None = None,
    ) -> None:
        """[RB-Y1 상태 고주파 구독 시작]

        공식 ``rby1_sdk``의 ``robot.start_state_update(cb, rate)`` 래퍼.
        ``on_state``는 ``rate`` Hz로 호출되며 인자는 1개(``robot_state``) 또는
        2개(``robot_state, control_manager_state``)를 받을 수 있다.

        콜백에 넘어오는 인자는 **raw pybind 객체**다. dict로 변환하려면 콜백 안에서
        ``RBRby1StateSDK.robot_state_to_dict(...)`` /
        ``RBRby1StateSDK.control_manager_state_to_dict(...)`` 를 호출할 것.

        주의: ``on_state`` 콜백은 gRPC 핸들러 **전용 스레드**에서 호출된다 —
        블로킹/무거운 작업 금지. asyncio 루프로 값을 넘기려면
        ``loop.call_soon_threadsafe`` 등으로 직접 브릿지할 것.
        """
        await asyncio.to_thread(
            self._sync_start_state_stream, robot_model, on_state, rate
        )
        self._finish_flow(flow_manager_args=flow_manager_args)

[문서]
    async def is_emergency_stopped(
        self,
        *,
        robot_model: str,
        flow_manager_args: FlowManagerArgs | None = None,
    ) -> bool:
        """[RB-Y1 비상정지(EMO) 상태 조회] EMO 버튼 중 하나라도 Pressed 면 True.

        ``get_state()``의 ``emo_states``(버튼별 ``Pressed``/``Released``)를 그대로
        씀. 물리 EMO 버튼이 없는 환경(예: 시뮬레이터)에서는 리스트가 비어 있어
        항상 False.
        """
        state = await self.get_state(robot_model=robot_model)
        emo_states = state.get("emo_states") or []
        result = any(e.get("state") == "Pressed" for e in emo_states)
        self._finish_flow(flow_manager_args=flow_manager_args)
        return result





[문서]
    async def is_connected(
        self,
        *,
        robot_model: str,
        flow_manager_args: FlowManagerArgs | None = None,
    ) -> bool:
        """[RB-Y1 연결 상태 조회]

        ``client.connected``는 C++ 호출일 수 있어 일관성을 위해 to_thread로 위임.
        """
        client = self.get_client(robot_model)
        try:
            connected = await asyncio.to_thread(lambda: client.connected)
        finally:
            self._finish_flow(flow_manager_args=flow_manager_args)
        return connected
