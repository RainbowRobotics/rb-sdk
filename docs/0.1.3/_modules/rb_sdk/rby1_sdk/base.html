rb_sdk.rby1_sdk.base의 소스 코드

"""RB-Y1 SDK 베이스 — 후보 model 전부에 대해 ``RBY1ClientSDK``를 미리 connect.

RB-Y1은 사이트당 1대 운용이지만 model이 ``a/m/ub`` 중 무엇인지 모를 수 있다.
부팅 시점에 후보 model 전부로 ``create_robot()`` + ``connect()``를 시도해
``_clients`` dict에 넣어두고, 함수 호출 시 들어오는 ``robot_model``로 lookup한다.
잘못된 model로 만들어진 client에 method를 호출하면 서버 측에서 no reply/에러가
나므로 자연스럽게 걸러진다.

비동기 정책:
    공식 ``rby1_sdk`` (PyPI 0.9.1)는 C++/pybind11 + sync gRPC stub이라
    async 컨텍스트에서 그냥 호출하면 이벤트 루프가 막힌다. 따라서 SDK의
    public 호출 메서드는 모두 ``async def``로 정의하고 내부에서
    ``asyncio.to_thread`` 로 worker thread에 위임한다. 자세한 근거는
    ``backend/packages/rb_sdk/src/rb_sdk/rby1_sdk/README.md`` 참조.
"""

# ----------------------------------------------------------------------
# 공식 rby1_sdk pybind 구조체(RobotState 등) → 순수 dict 변환 엔진.
#   pybind 객체는 __dict__/__slots__ 가 없어 generic reflection(t_to_dict/vars)이
#   실패하고, dir() 전수조사는 timedelta.max 같은 자기참조 상수로 무한재귀에 빠진다.
#   따라서 TypedDict 스키마에 선언된 필드만 따라 안전하게(범위 한정) 변환한다.
#   control/state 등 어디서든 ``RBRby1BaseSDK.struct_to_dict(obj, SomeTD)`` 로 재사용.
# ----------------------------------------------------------------------
_PRIMITIVES = (str, int, float, bool, datetime)


@functools.cache
def _hints(td: Any) -> dict[str, Any]:
    return get_type_hints(td)


def _vec(v: Any) -> Any:
    """numpy ndarray / sequence → 순수 파이썬 list (numpy import 없이)."""
    if v is None:
        return None
    if hasattr(v, "tolist"):
        return v.tolist()
    try:
        return list(v)
    except TypeError:
        return v


def _unopt(ann: Any) -> Any:
    """``X | None`` / ``Optional[X]`` → ``X``."""
    if get_origin(ann) in (Union, types.UnionType):
        rest = [a for a in get_args(ann) if a is not type(None)]
        if len(rest) == 1:
            return rest[0]
    return ann


def _obj_to_dict(o: Any) -> dict[str, Any]:
    """스키마 없는 객체(예: CollisionResult) → public 속성 dict (안전하게)."""
    out: dict[str, Any] = {}
    for k in dir(o):
        if k.startswith("_"):
            continue
        try:
            v = getattr(o, k)
        except Exception:  # noqa: BLE001 — 일부 바인딩 속성은 접근 시 raise 가능
            continue
        if callable(v):
            continue
        out[k] = _vec(v) if hasattr(v, "tolist") else v
    return out


def _convert(raw: Any, ann: Any) -> Any:
    if raw is None:
        return None
    ann = _unopt(ann)
    origin = get_origin(ann)

    if is_typeddict(ann):
        return _struct_to_dict(raw, ann)

    if origin in (list, tuple):
        args = get_args(ann)
        elem = _unopt(args[0]) if args else Any
        # 숫자 벡터/행렬은 통째로 tolist (원소별 변환보다 빠름)
        if elem in (float, int, bool, str) or get_origin(elem) in (list, tuple):
            return _vec(raw)
        return [_convert(x, elem) for x in raw]

    if origin is dict:
        return raw if isinstance(raw, dict) else _obj_to_dict(raw)

    # 스칼라
    if isinstance(raw, timedelta):  # rby1_sdk time_since_last_update 등 → 초(float)
        return raw.total_seconds()
    if isinstance(raw, _PRIMITIVES):
        return raw
    if hasattr(raw, "tolist"):  # numpy scalar/array
        return raw.tolist()
    if hasattr(raw, "name"):  # pybind enum
        return raw.name
    return raw


def _struct_to_dict(obj: Any, td: Any) -> dict[str, Any] | None:
    if obj is None:
        return None
    return {f: _convert(getattr(obj, f, None), ann) for f, ann in _hints(td).items()}

class RBRby1BaseSDK(RBBaseSDK):
    # 연결 풀은 endpoint 까지 포함한 (endpoint, model) 로 키잉한다.
    # 한 프로세스에서 서로 다른 endpoint(=다른 로봇)를 동시에 다뤄도
    # 풀이 섞이지 않도록 endpoint 별로 분리한다.
    _clients: ClassVar[dict[tuple[str, str], RBY1ClientSDK]] = {}
    _client_lock: ClassVar[threading.Lock] = threading.Lock()

    @classmethod
    def _singleton_subkey(cls, endpoint: str | None = None, *args: Any, **kwargs: Any) -> Any:
        # endpoint 가 다르면 다른 로봇 → state/control sub-SDK 도 endpoint 별 싱글톤으로
        # 분리해야 각 인스턴스가 자기 endpoint 의 client 만 보게 된다.
        return (endpoint,)

    def __init__(self, endpoint: str | None = None) -> None:
        super().__init__()
        # 싱글톤이 endpoint 로 분리되므로 같은 인스턴스는 항상 같은 endpoint.
        self._endpoint = endpoint

    @classmethod
    def configure(
        cls,
        config: RBY1Config,
        *,
        candidates: tuple[str, ...] = DEFAULT_CANDIDATES,
        cancel_event: threading.Event | None = None,
    ) -> None:
        """``config.host:port``로 후보 model 전부 (또는 ``config.model``만) connect.

        - ``config.model``이 명시되면 그 1개만
        - 명시 안 됐으면 ``candidates`` (기본 ``a, m, ub``) 전부 시도
        - connect 실패한 model은 dict에 넣지 않음 (warning만 남김)
        - 멱등 — 이미 등록된 model은 재시도하지 않음
        - ``cancel_event``가 set 되면 진행 중인 blocking connect가 끝난 직후 결과를 폐기
        """
        models = [config.model] if config.model else list(candidates)
        endpoint = config.endpoint

        with cls._client_lock:
            pending_models = [
                model for model in models if (endpoint, model) not in cls._clients
            ]

        if not pending_models:
            return

        def _probe_model(model: str) -> tuple[str, RBY1ClientSDK | None, str, Exception | None]:
            if cancel_event is not None and cancel_event.is_set():
                return model, None, "cancelled", None

            cfg = RBY1Config(
                endpoint=endpoint,
                host=config.host,
                port=config.port,
                model=model,
            )
            client = RBY1ClientSDK(cfg)
            try:
                client.connect()
            except Exception as e:  # noqa: BLE001
                if cancel_event is not None and cancel_event.is_set():
                    with contextlib.suppress(Exception):
                        client.disconnect()
                    return model, None, "cancelled", None
                return model, client, "failed", e

            if cancel_event is not None and cancel_event.is_set():
                with contextlib.suppress(Exception):
                    client.disconnect()
                return model, None, "cancelled", None

            # 소켓 connect 는 후보 model 전부 성공한다(같은 로봇으로 채널만 열림).
            # 실제 model 과 다른 Robot_* 클래스는 state 를 읽을 때 DOF mismatch 로
            # 깨지므로, connect 직후 state 를 한 번 읽어 진짜 model 만 남긴다.
            try:
                cls._validate_model(client)
            except Exception as e:  # noqa: BLE001
                if cancel_event is not None and cancel_event.is_set():
                    with contextlib.suppress(Exception):
                        client.disconnect()
                    return model, None, "cancelled", None
                return model, client, "excluded", e

            return model, client, "connected", None

        with concurrent.futures.ThreadPoolExecutor(
            max_workers=len(pending_models),
            thread_name_prefix=f"rby1-probe-{endpoint}",
        ) as executor:
            futures = [executor.submit(_probe_model, model) for model in pending_models]
            was_cancelled = False

            for future in concurrent.futures.as_completed(futures):
                model, client, status, error = future.result()

                if status == "cancelled":
                    was_cancelled = True
                    continue

                if cancel_event is not None and cancel_event.is_set():
                    was_cancelled = True
                    if client is not None:
                        with contextlib.suppress(Exception):
                            client.disconnect()
                    continue

                if status == "failed":
                    if client is not None:
                        with contextlib.suppress(Exception):
                            client.disconnect()
                    continue

                if status == "excluded":
                    if client is not None:
                        with contextlib.suppress(Exception):
                            client.disconnect()
                    continue

                if client is None:
                    continue

                with cls._client_lock:
                    if cancel_event is not None and cancel_event.is_set():
                        with contextlib.suppress(Exception):
                            client.disconnect()
                        return
                    if (endpoint, model) in cls._clients:
                        with contextlib.suppress(Exception):
                            client.disconnect()
                        continue
                    cls._clients[(endpoint, model)] = client

            if was_cancelled:
                return

        if cancel_event is not None and cancel_event.is_set():
            with cls._client_lock:
                stale_clients = [
                    key for key in cls._clients if key[0] == endpoint and key[1] in pending_models
                ]
                for key in stale_clients:
                    client = cls._clients.pop(key)
                    with contextlib.suppress(Exception):
                        client.disconnect()



    @staticmethod
    def _validate_model(client: RBY1ClientSDK) -> None:
        """후보 model 이 로봇의 실제 model 과 일치하는지 state 읽기로 검증.

        잘못된 model 로 만든 client 는 ``get_state`` 에서 "size of joint state does
        not match the DOF" 로 raise 한다. 일치하면 정상 반환, 불일치면 예외 전파.
        부팅 직후 state 가 아직 안 올라온 경우를 대비해 짧게 재시도한다.
        """
        robot = client.robot
        if robot is None:
            raise RBY1SDKConnectionError("rby1_not_connected")

        fn = next(
            (
                f
                for name in ("get_state", "GetState", "state")
                if callable(f := getattr(robot, name, None))
            ),
            None,
        )
        if fn is None:
            # state 메서드가 없으면 검증 수단이 없으므로 보수적으로 통과시킨다.
            return

        last_exc: Exception | None = None
        for _ in range(5):
            try:
                fn()
                return
            except Exception as e:  # noqa: BLE001 — DOF mismatch 포함, 재시도 후 전파
                last_exc = e
                time.sleep(0.2)
        if last_exc is not None:
            raise last_exc

    @classmethod
    def reset(cls) -> None:
        """테스트/재설정용. 등록된 모든 client disconnect 후 레지스트리 비움."""
        with cls._client_lock:
            for client in cls._clients.values():
                with contextlib.suppress(Exception):
                    client.disconnect()
            cls._clients.clear()





[문서]
    @classmethod
    def reset_endpoint(cls, endpoint: str | None) -> None:
        """특정 endpoint 에 등록된 client 만 disconnect 후 레지스트리에서 제거."""
        with cls._client_lock:
            keys = [key for key in cls._clients if key[0] == endpoint]
            for key in keys:
                client = cls._clients.pop(key)
                with contextlib.suppress(Exception):
                    client.disconnect()

    def get_client(self, robot_model: str) -> RBY1ClientSDK:
        """이 인스턴스의 endpoint + ``robot_model`` 에 매핑된 client 반환. 없으면 예외."""
        cls = type(self)
        endpoint = getattr(self, "_endpoint", None)
        with cls._client_lock:
            client = cls._clients.get((endpoint, robot_model))
        if client is None:
            registered = [m for (ep, m) in cls._clients if ep == endpoint]
            raise RBY1SDKConnectionError(
                f"rby1 client not registered for endpoint={endpoint!r} "
                f"model={robot_model!r} (registered: {registered})"
            )
        return client

    def get_robot(self, robot_model: str) -> Any:
        """공식 ``rby1_sdk``의 Robot 인스턴스 (Robot_A / Robot_M / Robot_UB)."""
        return self.get_client(robot_model).robot



    @property
    def models(self) -> list[str]:
        """이 인스턴스의 endpoint 에 connect 된 model 키 목록."""
        endpoint = getattr(self, "_endpoint", None)
        return [m for (ep, m) in type(self)._clients if ep == endpoint]

    @staticmethod
    def struct_to_dict(obj: Any, schema: Any) -> dict[str, Any] | None:
        """공식 ``rby1_sdk`` pybind 구조체 → 순수 dict.

        ``schema`` (TypedDict) 에 선언된 필드만 따라 안전하게(재귀 범위 한정) 변환한다.
        numpy ndarray → list, datetime 유지, timedelta → 초(float), pybind enum → name.
        중첩 TypedDict / 리스트도 재귀 처리.

        예시::

            data = RBRby1BaseSDK.struct_to_dict(robot.get_state(), RobotStateTD)
        """
        return _struct_to_dict(obj, schema)

[문서]
    async def call(
        self,
        robot_model: str,
        method: str,
        *args: Any,
        flow_manager_args: FlowManagerArgs | None = None,
        **kwargs: Any,
    ) -> Any:
        """공식 robot 객체의 임의 메서드 호출 (sync gRPC를 worker thread로 위임)."""
        client = self.get_client(robot_model)
        result = await asyncio.to_thread(client.call, method, *args, **kwargs)
        self._finish_flow(flow_manager_args=flow_manager_args)
        return result





[문서]
    async def whoami(
        self,
        robot_model: str,
        *,
        flow_manager_args: FlowManagerArgs | None = None,
    ) -> dict[str, Any]:
        """``robot.get_robot_info()`` 호출."""
        client = self.get_client(robot_model)
        result = await asyncio.to_thread(client.whoami)
        self._finish_flow(flow_manager_args=flow_manager_args)
        return result



    @staticmethod
    def _finish_flow(*, flow_manager_args: FlowManagerArgs | None) -> None:
        if flow_manager_args is not None:
            flow_manager_args.done()

    @staticmethod
    def run_sync(coro: Any) -> Any:
        """sync 컨텍스트(PyFM Step 등)에서 ``async def`` 메서드를 호출하기 위한 헬퍼.

        예시::

            from rb_sdk import RBRby1SDK
            rby1 = RBRby1SDK()
            result = RBRby1BaseSDK.run_sync(
                rby1.control.power_on(robot_model="a")
            )
        """
        try:
            asyncio.get_running_loop()
        except RuntimeError:
            return asyncio.run(coro)
        raise RuntimeError(
            "run_sync called from within a running event loop. "
            "Use `await coroutine` directly instead."
        )
