from __future__ import annotations

import asyncio
import contextlib
import threading
from typing import Any

from rb_log.log import rb_log

from .base import RBBaseSDK
from .rby1_sdk.base import RBRby1BaseSDK
from .rby1_sdk.client import RBY1ClientSDK
from .rby1_sdk.config import RBY1Config
from .rby1_sdk.control import RBRby1ControlSDK
from .rby1_sdk.state import RBRby1StateSDK

# 최초 연결 시도(또는 네트워크 장애 등으로 인한 연결 끊김) 후 로봇이 안 붙어 있으면
# 이 간격으로 계속 재시도한다. configure()는 이미 연결된 model은 건너뛰므로(멱등)
# 반복 호출해도 안전하다.
RECONNECT_RETRY_INTERVAL_SEC = 5.0

class RBRby1SDK(RBBaseSDK):
    """Rainbow Robotics RB-Y1 Total SDK

    RB-Y1은 사이트당 1대 운용이지만 model(``a`` / ``m`` / ``ub``)이 무엇인지
    모를 수 있다. 인스턴스화 시점에 후보 model 전부로 connect를 시도해 dict에
    넣어두고, 함수 호출 시 들어오는 ``robot_model``로 lookup한다.

    잘못된 model 클래스로 만들어진 client에 method를 호출하면 서버 쪽에서
    no reply/에러가 나므로 자연스럽게 걸러진다.

    예시:
        rby1 = RBRby1SDK()                                  # 127.0.0.1:50051, a/m/ub 전부 시도
        await rby1.connect(host="192.168.1.50")             # 외부 host 로 재연결
        rby1 = RBRby1SDK(RBY1Config(host="192.168.1.50"))   # 생성 시 host 지정도 가능
        rby1 = RBRby1SDK(RBY1Config(model="a"))             # model 지정 → a만 connect

        rby1.control.power_on(robot_model="a")              # path param 그대로 라우팅
    """

    control: RBRby1ControlSDK
    state: RBRby1StateSDK

    def __new__(cls, *args, **kwargs):
        # config(endpoint/model)가 다르면 다른 로봇 → 별도 싱글톤으로 분리해야 한다.
        # subkey 계산을 위해 생성 인자를 base.__new__ 로 그대로 넘긴다.
        return super().__new__(cls, *args, **kwargs)

    @staticmethod
    def _has_connection_kwargs(
        config: RBY1Config | None = None,
        *,
        endpoint: str | None = None,
        host: str | None = None,
        port: int | None = None,
        model: str | None = None,
    ) -> bool:
        return config is not None or any(v is not None for v in (endpoint, host, port, model))

    @staticmethod
    def _resolve_config(
        config: RBY1Config | None = None,
        *,
        endpoint: str | None = None,
        host: str | None = None,
        port: int | None = None,
        model: str | None = None,
    ) -> RBY1Config:
        if config is not None and any(v is not None for v in (endpoint, host, port, model)):
            raise ValueError("Pass either `config` or endpoint/host/port/model keywords, not both.")

        if config is not None:
            return config

        kwargs: dict[str, Any] = {
            "endpoint": endpoint,
            "model": model,
        }
        if host is not None:
            kwargs["host"] = host
        if port is not None:
            kwargs["port"] = port
        return RBY1Config(**kwargs)

    @classmethod
    def _singleton_subkey(
        cls,
        config: RBY1Config | None = None,
        *,
        endpoint: str | None = None,
        host: str | None = None,
        port: int | None = None,
        model: str | None = None,
    ) -> Any:
        cfg = cls._resolve_config(
            config,
            endpoint=endpoint,
            host=host,
            port=port,
            model=model,
        )
        return (cfg.endpoint, cfg.model)

    def __init__(
        self,
        config: RBY1Config | None = None,
        *,
        endpoint: str | None = None,
        host: str | None = None,
        port: int | None = None,
        model: str | None = None,
    ):
        already_initialized = getattr(self, "_initialized", False)
        super().__init__()

        if already_initialized and not self._has_connection_kwargs(
            config,
            endpoint=endpoint,
            host=host,
            port=port,
            model=model,
        ):
            return

        cfg = self._resolve_config(
            config,
            endpoint=endpoint,
            host=host,
            port=port,
            model=model,
        )
        self._endpoint = cfg.endpoint

        # sub-SDK 도 endpoint 별 싱글톤 — 자기 endpoint 의 client 만 보게 한다.
        if not hasattr(self, "control"):
            self.control = RBRby1ControlSDK(endpoint=self._endpoint)
        if not hasattr(self, "state"):
            self.state = RBRby1StateSDK(endpoint=self._endpoint)

        if not hasattr(self, "_pending_configure_cancel"):
            self._pending_configure_cancel: threading.Event | None = None
            self._pending_configure_thread: threading.Thread | None = None

        # 생성자는 이벤트 루프/서비스 startup 을 막지 않도록 백그라운드에서 probing 한다.
        self._start_configure_in_background(cfg)

    def _cancel_pending_configure(self) -> None:
        cancel = getattr(self, "_pending_configure_cancel", None)
        if cancel is not None:
            cancel.set()

    def _start_configure_in_background(self, cfg: RBY1Config) -> None:
        self._cancel_pending_configure()
        cancel = threading.Event()
        self._pending_configure_cancel = cancel

        def _worker() -> None:
            try:
                while not cancel.is_set():
                    try:
                        self._prune_disconnected_clients()
                        # 하나라도 연결돼 있으면 이번 tick은 건드리지 않는다. 전부
                        # 끊긴 상태일 때만(최초 연결이든, 붙어있다 나중에 다 끊긴
                        # 것이든) 후보 model 전부를 병렬로 다시 probe 한다.
                        if not self.models:
                            RBRby1BaseSDK.configure(cfg, cancel_event=cancel)
                    except Exception as e:  # noqa: BLE001 — 워치독은 절대 죽으면 안 됨
                        rb_log.error(f"[RBRby1SDK] reconnect watchdog tick failed: {e}")
                    if cancel.is_set():
                        return
                    cancel.wait(RECONNECT_RETRY_INTERVAL_SEC)
            finally:
                if getattr(self, "_pending_configure_cancel", None) is cancel:
                    self._pending_configure_cancel = None

        thread = threading.Thread(
            target=_worker,
            daemon=True,
            name=f"rby1-connect-{cfg.endpoint}",
        )
        self._pending_configure_thread = thread
        thread.start()

    def _prune_disconnected_clients(self) -> None:
        """이 endpoint에 등록된 client 중 실제로 끊긴 걸 풀에서 제거한다.

        ``configure()``는 이미 ``_clients``에 있는 (endpoint, model)은 건드리지
        않으므로, 살아있는 줄 알고 등록해둔 client가 실제로는 끊겼을 때(로봇 재부팅,
        네트워크 단절 등) 이렇게 직접 걷어내야 다음 tick에 재연결 대상으로 다시 잡힌다.
        """
        endpoint = self._endpoint

        def _is_dead(client: RBY1ClientSDK) -> bool:
            try:
                return not client.connected
            except Exception:  # noqa: BLE001 — 판정 불가면 죽었다고 보지 않는다(보수적)
                return False

        removed: list[RBY1ClientSDK] = []
        with RBRby1BaseSDK._client_lock:
            stale_keys = [
                key
                for key, client in RBRby1BaseSDK._clients.items()
                if key[0] == endpoint and _is_dead(client)
            ]
            for key in stale_keys:
                removed.append(RBRby1BaseSDK._clients.pop(key))

        for client in removed:
            with contextlib.suppress(Exception):
                client.disconnect()

    @property
    def models(self) -> list[str]:
        """이 endpoint 에 connect 된 model 키 목록."""
        endpoint = self._endpoint
        return [m for (ep, m) in RBRby1BaseSDK._clients if ep == endpoint]

    @property
    def is_connecting(self) -> bool:
        """백그라운드 RB-Y1 connect/probe 작업이 아직 실행 중인지 여부."""
        thread = getattr(self, "_pending_configure_thread", None)
        return bool(thread is not None and thread.is_alive())

    async def wait_until_connected(
        self,
        *,
        timeout: float = 10.0,
        interval: float = 0.1,
    ) -> list[str]:
        """백그라운드 probing 이 끝나거나 model 이 등록될 때까지 non-blocking 대기."""
        loop = asyncio.get_running_loop()
        deadline = loop.time() + timeout

        while True:
            models = self.models
            if models:
                return models
            if not self.is_connecting:
                return []
            if loop.time() >= deadline:
                return []
            await asyncio.sleep(min(interval, max(0.0, deadline - loop.time())))

    async def connect(
        self,
        *,
        host: str | None = None,
        port: int | None = None,
        endpoint: str | None = None,
        model: str | None = None,
    ) -> list[str]:
        """RB-Y1 gRPC endpoint 를 재설정한다.

        ``await sdk.connect(host="...")`` 로 다른 로봇/PC 로 재연결하면 현재
        endpoint client pool 을 닫고 새 endpoint 로 다시 connect 한다.
        """
        cfg = self._resolve_config(
            endpoint=endpoint,
            host=host,
            port=port,
            model=model,
        )

        self._cancel_pending_configure()
        old_endpoint = getattr(self, "_endpoint", None)
        RBRby1BaseSDK.reset_endpoint(old_endpoint)
        if cfg.endpoint != old_endpoint:
            RBRby1BaseSDK.reset_endpoint(cfg.endpoint)

        self._endpoint = cfg.endpoint
        self.control = RBRby1ControlSDK(endpoint=self._endpoint)
        self.state = RBRby1StateSDK(endpoint=self._endpoint)
        await asyncio.to_thread(RBRby1BaseSDK.configure, cfg)
        # 지금 시도가 실패하거나(로봇이 아직 준비 안 됨) 나중에 끊기더라도 계속
        # 재연결하도록, 이후는 __init__ 때와 동일한 백그라운드 watchdog에 맡긴다.
        self._start_configure_in_background(cfg)
        return self.models

    async def disconnect(self) -> bool:
        """현재 RB-Y1 gRPC endpoint client pool 을 닫는다."""
        self._cancel_pending_configure()
        endpoint = getattr(self, "_endpoint", None)
        had_clients = any(ep == endpoint for ep, _ in RBRby1BaseSDK._clients)
        RBRby1BaseSDK.reset_endpoint(endpoint)
        return had_clients
