    def parse_setting_param_type(self, type: int) -> SettingParamTypeEnum:
        if type == 0:
            return SettingParamTypeEnum.STRING
        elif type == 1:
            return SettingParamTypeEnum.INT
        elif type == 2:
            return SettingParamTypeEnum.FLOAT
        elif type == 3:
            return SettingParamTypeEnum.BOOL
        else:
            raise RuntimeError("Invalid setting param type")
